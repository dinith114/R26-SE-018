# Dependencies for the CI test run only.
#
# Deliberately NOT requirements.txt. That file pulls TensorFlow and Torch -
# roughly 7 GB including CUDA wheels the runner has no GPU for - which would
# make every pull request take ten minutes to tell you about a typo.
#
# The tests exercise pure decision logic: freshness windows, the farm clock, the
# auto/manual switch and the alarm repeat rule. None of them run a model, so
# none of them need the ML stack. What they DO need is whatever
# smart_care_v2.py and automation.py touch at import time.
#
# The full dependency set is still checked, by the `deps` job, which resolves
# requirements.txt without downloading it. That is what catches a conflict like
# the duplicate tensorflow pin.

fastapi==0.115.0
pydantic==2.9.0
pydantic-settings==2.5.0
numpy==1.26.4
joblib==1.4.2
scikit-learn==1.5.0
requests==2.32.3
pytest==8.3.0
# smart_care_v2 imports its Firebase helpers from smart_watering, which uses
# pandas and declares a Form/File upload route - so both are needed to reach
# the first assert, even though no test uploads anything.
pandas==2.2.2
python-multipart==0.0.9
PyKrige==1.7.2
# Sparse sensor placement (Brunton lab). The PyPI name is python-sensors:
# "pysensors" is an unrelated Linux lm-sensors binding that fails to build.
# Imported as `pysensors`. Documented for Linux and macOS only, so house_planner
# degrades to its kriging-variance method rather than failing when absent.
# 0.4.1, NOT 0.4.3: from 0.4.2 the package requires numpy>=2.0, which
# collides with the numpy==1.26.4 this project pins for its saved models.
# 0.4.1 leaves numpy unpinned and carries the same SSPOR / SVD API.
python-sensors==0.4.1
# fastapi.testclient needs httpx. The accounts and guard tests drive real
# routing through TestClient rather than calling the dependencies directly.
httpx==0.27.0
# app/main.py imports uvicorn at module load, and test_main_wiring.py has to
# import app.main - the thing it tests is main.py's static-key middleware
# sitting in front of the accounts routes, which every other test misses
# because they all build a bare FastAPI() and mount the router themselves.
uvicorn==0.30.6
